#include <iostream>
#include "blank.hpp"
#include <string>
#include <vector>

int main()
{
	setlocale(LC_ALL, "Russian");
	mt::Blank blank1;
	blank1.printInfo();
	std::cout << "-------------------" << std::endl;
	std::cout << "Использование геттеров:" << std::endl;
	std::cout << "Имя: " << blank1.get_name() << std::endl;
	std::cout << "Фамилия: " << blank1.get_surname() << std::endl;
	std::cout << "Долги: ";
	std::vector<std::string> dolgs1 = blank1.get_dolgs();
	for (std::vector<std::string>::size_type i = 0; i < dolgs1.size(); ++i) {
		std::cout << dolgs1[i] << " ";
	}
	std::cout << std::endl;

	std::cout << "-------------------Студенты с долгами-------------------" << std::endl;
	std::vector<std::string> dolgs2 = { "matanalyz", "istoria", "programmirovanie" };
	mt::Blank blank2("leonid", "pechevisty", dolgs2);
	std::cout << "\nblank2:" << std::endl;
	blank2.printInfo();

	std::vector<std::string> dolgs3 = { "matanalyz", "istoria", "algebra", "algebra"};
	mt::Blank blank3("anton", "ushakov", dolgs3);
	std::cout << "\nblank3:" << std::endl;
	blank3.printInfo();

	std::cout << "-------------------Оператор + (объединение долгов)-------------------" << std::endl;

	mt::Blank blank4 = blank2 + blank3;
	std::cout << "Результат blank2 + blank3:" << std::endl;
	blank4.printInfo();

	std::cout << "-------------------Оператор -= (вычитание долгов)-------------------" << std::endl;

	mt::Blank blank5 = blank2;
	std::cout << "До вычитания (blank5 - blank2):" << std::endl;
	blank5.printInfo();

	blank5 -= blank3;
	blank5.printInfo();

	std::cout << "-------------------Оператор / (общие долги)-------------------" << std::endl;

	mt::Blank blank6 = blank2 / blank3;
	std::cout << "Результат blank2 / blank3:" << std::endl;
	blank6.printInfo();

	std::cout << "-------------------Оператор присваивания-------------------" << std::endl;

	mt::Blank blank7;
	std::cout << "blank7 до присваивания:" << std::endl;
	blank7.printInfo();

	blank7 = blank2;
	blank7.printInfo();

	std::cout << "-------------------Добавление долга-------------------" << std::endl;

	blank1.add_dolg("angliyskiy");
	std::cout << "результат:" << std::endl;
	blank1.printInfo();

	std::cout << "-------------------Вывод неправильных данных-------------------" << std::endl;
	try {
		blank1.set_name("");
	}
	catch (const std::exception& ex) {
		std::cerr << "Ошибка: " << ex.what() << std::endl;
	}
	try {
		blank1.add_dolg("");
	}
	catch (const std::exception& ex) {
		std::cerr << "Ошибка: " << ex.what() << std::endl;
	}

	return 0;
}
